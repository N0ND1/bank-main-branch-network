 description DOWNLINK to ASW1-ASW6
! skip the next line if the platform rejects it (dot1q only)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,40,50,60,70,80,90,99,100
 switchport nonegotiate
 no shutdown
!
interface range gigabitEthernet1/0/23 - 24
 description Po1 member to L3SW2
 no switchport

 no shutdown
!
interface Port-channel1
 description L3 EtherChannel to L3SW2
 no switchport
 ip address 10.255.255.1 255.255.255.252
 no shutdown
!
interface GigabitEthernet1/0/1
 description EDGE to Router0 Gi0/0/1

 ip address 10.10.255.2 255.255.255.252
 no shutdown

! ----- base, VLANs, trunks, Po1 -----
enable
configure terminal
hostname L3SW2
ip routing
!
vlan 10
 name MANAGEMENT
vlan 20
 name FINANCE
vlan 30
 name HR
vlan 40
 name IT
vlan 50

vlan 60

vlan 70

vlan 80

vlan 90

vlan 99

vlan 100

!
! Gi1/0/3 ASW1 | /4 ASW2 | /5 ASW3 | /6 ASW4 | /7 ASW5 | /8 ASW6
interface range gigabitEthernet1/0/3 - 8
 description DOWNLINK to ASW1-ASW6
! skip the next line if the platform rejects it (dot1q only)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,40,50,60,70,80,90,99,100
 switchport nonegotiate
 no shutdown
!
interface range gigabitEthernet1/0/23 - 24
 description Po1 member to L3SW1
 no switchport
 channel-group 1 mode active
 no shutdown
!
interface Port-channel1
 description L3 EtherChannel to L3SW1
 no switchport
 ip address 10.255.255.2 255.255.255.252
 no shutdown
!
! Gi1/0/1 unused on L3SW2 (edge link is on L3SW1)
!
